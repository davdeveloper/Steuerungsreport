function postNewsToChat(team, title, beschreibung, bild, teilbarerlink) {
  const webhookUrl = PropertiesService
    .getScriptProperties()
    .getProperty('CHAT_WEBHOOK_URL');

  if (!webhookUrl) throw new Error('CHAT_WEBHOOK_URL fehlt.');

  const escapeHtml = value => String(value ?? '')
    .replace(/&/g, '&amp;')
    .replace(/</g, '&lt;')
    .replace(/>/g, '&gt;')
    .replace(/"/g, '&quot;')
    .replace(/'/g, '&#39;');

  const link = String(teilbarerlink ?? '').trim();
  if (!link.startsWith('https://')) {
    throw new Error('Teilbarerlink muss eine vollständige HTTPS-URL sein.');
  }

  const widgets = [];
  const bildUrl = String(bild ?? '').trim();

  // Chat benötigt hier eine direkt erreichbare HTTPS-Bildadresse.
  if (bildUrl.startsWith('https://')) {
    widgets.push({
      image: { imageUrl: bildUrl, altText: 'News-Bild' }
    });
  }

  widgets.push({
    textParagraph: {
      text:
        `<b>📢 Es gibt News im Bereich ${escapeHtml(team)}</b>` +
        `<br><br><b>Thema:</b> ${escapeHtml(title)}` +
        `<br><br>${escapeHtml(beschreibung).replace(/\r\n|\r|\n/g, '<br>')}`
    }
  });

  widgets.push({
    buttonList: {
      buttons: [{
        text: 'News im Browser öffnen',
        onClick: { openLink: { url: link } }
      }]
    }
  });

  const response = UrlFetchApp.fetch(webhookUrl, {
    method: 'post',
    contentType: 'application/json; charset=UTF-8',
    payload: JSON.stringify({
      cardsV2: [{
        cardId: 'newsCard',
        card: { sections: [{ widgets }] }
      }]
    }),
    muteHttpExceptions: true
  });

  const status = response.getResponseCode();
  if (status < 200 || status >= 300) {
    throw new Error(`Google Chat HTTP ${status}: ${response.getContentText()}`);
  }
}
